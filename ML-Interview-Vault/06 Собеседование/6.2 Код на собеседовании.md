## Линейная регрессия: аналитически и градиентным спуском
```python
def linreg_closed_form(X, y, l2=0.0):
    X = np.c_[np.ones(len(X)), X]                       # bias-столбец
    I = np.eye(X.shape[1]); I[0, 0] = 0                 # bias не регуляризуем
    return np.linalg.solve(X.T @ X + l2 * I, X.T @ y)   # solve стабильнее, чем inv

def linreg_gd(X, y, lr=0.1, epochs=1000):
    n, d = X.shape
    w, b = np.zeros(d), 0.0
    for _ in range(epochs):
        err = X @ w + b - y                             # (n,)
        w -= lr * 2 / n * X.T @ err                     # ∇ MSE
        b -= lr * 2 / n * err.sum()
    return w, b
```

## Логистическая регрессия
```python
def sigmoid(z):
    return 1 / (1 + np.exp(-np.clip(z, -500, 500)))

def logreg_fit(X, y, lr=0.1, epochs=1000, l2=0.0):
    n, d = X.shape
    w, b = np.zeros(d), 0.0
    for _ in range(epochs):
        p = sigmoid(X @ w + b)
        w -= lr * (X.T @ (p - y) / n + l2 * w)          # ∇ = Xᵀ(p − y)/n
        b -= lr * (p - y).mean()
    return w, b

def log_loss(y, p, eps=1e-15):
    p = np.clip(p, eps, 1 - eps)
    return -np.mean(y * np.log(p) + (1 - y) * np.log(1 - p))
```

## Softmax и cross-entropy (численно стабильно)
```python
def softmax(z, axis=-1):
    z = z - z.max(axis=axis, keepdims=True)             # вычитаем max — не меняет результат
    e = np.exp(z)
    return e / e.sum(axis=axis, keepdims=True)

def cross_entropy(logits, y):                           # y — индексы классов
    z = logits - logits.max(axis=1, keepdims=True)
    log_probs = z - np.log(np.exp(z).sum(axis=1, keepdims=True))   # log-sum-exp
    return -log_probs[np.arange(len(y)), y].mean()
```

## kNN
```python
def knn_predict(X_train, y_train, X_test, k=5):
    # ‖a−b‖² = ‖a‖² + ‖b‖² − 2a·b  — векторизованно
    d = (X_test**2).sum(1)[:, None] + (X_train**2).sum(1)[None, :] - 2 * X_test @ X_train.T
    idx = np.argpartition(d, k, axis=1)[:, :k]          # k ближайших, O(n)
    return np.array([np.bincount(row).argmax() for row in y_train[idx]])
```

## K-Means
```python
def kmeans(X, k, iters=100, seed=0):
    rng = np.random.default_rng(seed)
    C = X[rng.choice(len(X), k, replace=False)]
    for _ in range(iters):
        labels = ((X[:, None, :] - C[None, :, :])**2).sum(-1).argmin(1)   # (n, k) → (n,)
        newC = np.array([X[labels == j].mean(0) if np.any(labels == j) else C[j]
                         for j in range(k)])
        if np.allclose(newC, C):
            break
        C = newC
    return C, labels
```

## Метрики
```python
def precision_recall_f1(y, y_pred):
    tp = np.sum((y == 1) & (y_pred == 1))
    fp = np.sum((y == 0) & (y_pred == 1))
    fn = np.sum((y == 1) & (y_pred == 0))
    p = tp / (tp + fp) if tp + fp else 0.0
    r = tp / (tp + fn) if tp + fn else 0.0
    f1 = 2 * p * r / (p + r) if p + r else 0.0
    return p, r, f1
```

### ROC-AUC
Через интерпретацию «P(score_pos > score_neg)» и ранги (Манн–Уитни), $O(n\log n)$:
```python
from scipy.stats import rankdata

def roc_auc(y, scores):
    ranks = rankdata(scores)                            # средние ранги для ничьих
    n_pos = y.sum(); n_neg = len(y) - n_pos
    return (ranks[y == 1].sum() - n_pos * (n_pos + 1) / 2) / (n_pos * n_neg)

def roc_auc_naive(y, s):                                # O(n_pos · n_neg) — для проверки
    pos, neg = s[y == 1], s[y == 0]
    return ((pos[:, None] > neg[None, :]).mean()
            + 0.5 * (pos[:, None] == neg[None, :]).mean())
```

## Решающий пень / критерий Джини
```python
def gini(y):
    _, counts = np.unique(y, return_counts=True)
    p = counts / counts.sum()
    return 1 - np.sum(p**2)

def best_split(x, y):
    best = (None, np.inf)
    for t in np.unique(x)[:-1]:
        left, right = y[x <= t], y[x > t]
        score = (len(left) * gini(left) + len(right) * gini(right)) / len(y)
        if score < best[1]:
            best = (t, score)
    return best                                          # (порог, взвешенный Gini)
```

## Scaled dot-product attention и Multi-Head Attention
```python
def attention(Q, K, V, mask=None):
    # Q: (..., n_q, d_k), K: (..., n_k, d_k), V: (..., n_k, d_v)
    scores = Q @ K.swapaxes(-1, -2) / np.sqrt(Q.shape[-1])     # (..., n_q, n_k)
    if mask is not None:
        scores = np.where(mask, scores, -1e9)
    w = softmax(scores, axis=-1)
    return w @ V, w

def causal_mask(n):
    return np.tril(np.ones((n, n), dtype=bool))

def multi_head_attention(X, Wq, Wk, Wv, Wo, h, causal=False):
    n, d = X.shape
    dk = d // h
    def split(M):                                       # (n, d) → (h, n, dk)
        return M.reshape(n, h, dk).transpose(1, 0, 2)
    Q, K, V = split(X @ Wq), split(X @ Wk), split(X @ Wv)
    mask = causal_mask(n) if causal else None
    out, _ = attention(Q, K, V, mask)                   # (h, n, dk)
    out = out.transpose(1, 0, 2).reshape(n, d)          # concat голов
    return out @ Wo
```

## Мини-трансформер-блок на PyTorch
```python
import torch, torch.nn as nn, torch.nn.functional as F

class Block(nn.Module):
    def __init__(self, d=256, h=4, ff=4):
        super().__init__()
        self.h = h
        self.ln1, self.ln2 = nn.LayerNorm(d), nn.LayerNorm(d)
        self.qkv, self.proj = nn.Linear(d, 3 * d), nn.Linear(d, d)
        self.mlp = nn.Sequential(nn.Linear(d, ff * d), nn.GELU(), nn.Linear(ff * d, d))

    def attn(self, x):
        B, T, C = x.shape
        q, k, v = self.qkv(x).split(C, dim=2)
        q, k, v = (t.view(B, T, self.h, C // self.h).transpose(1, 2) for t in (q, k, v))
        y = F.scaled_dot_product_attention(q, k, v, is_causal=True)   # FlashAttention внутри
        return self.proj(y.transpose(1, 2).reshape(B, T, C))

    def forward(self, x):                               # Pre-LN
        x = x + self.attn(self.ln1(x))
        x = x + self.mlp(self.ln2(x))
        return x
```

## LoRA-слой
```python
class LoRALinear(nn.Module):
    def __init__(self, base: nn.Linear, r=8, alpha=16):
        super().__init__()
        self.base = base
        for p in self.base.parameters():
            p.requires_grad = False                     # заморозили W₀
        self.A = nn.Parameter(torch.randn(r, base.in_features) * 0.01)
        self.B = nn.Parameter(torch.zeros(base.out_features, r))   # B=0 → ΔW=0 на старте
        self.scale = alpha / r

    def forward(self, x):
        return self.base(x) + (x @ self.A.T @ self.B.T) * self.scale
```
